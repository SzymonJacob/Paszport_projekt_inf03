            {
                // Próba załadowania zdjęcia na podstawie numeru
                pictureBox1.Image = System.Drawing.Image.FromFile($"{basePath}{fileName}-zdjecie.jpg");

                pictureBox2.Image = System.Drawing.Image.FromFile($"{basePath}{fileName}-odcisk.jpg");

            {

                MessageBox.Show("ZLY NUMER!!!\nWpisz numer 000,111 czy 333");
            }
